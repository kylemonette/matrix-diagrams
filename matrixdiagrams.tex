\begin{center}
\begin{tabular}{l@{\qquad}l}
$  \matrixbrace[3]{\mbSpike{0.5}{0.5}{tl}}
$
&
\verb|\matrixbrace[3]{\mbSpike{0.5}{0.5}{tl}}| \\ [3em]
$  \matrixbrace[3]{\mbSpike{0.5}{0.5}{tr}}
$
&
\verb|\matrixbrace[3]{\mbSpike{0.5}{0.5}{tr}}| \\ [3em]
$  \matrixbrace[3]{\mbSpike{0.5}{0.5}{bl}}
$
&
\verb|\matrixbrace[3]{\mbSpike{0.5}{0.5}{bl}}| \\ [3em]

\section{Command reference}
% ===================================================================

\begin{center}
\begin{tabular}{ll}
\toprule
\verb|\matrixbrace[s][a]{c}|
  & Main command. \texttt{s} = scale (def.\ 1), \texttt{a} = aspect (def.\ 1 = square). \\
\midrule
\verb|\mbDiag|
  & Anti-diagonal $(0,H)\to(1,0)$. \\
\verb|\mbDiagBand|
  & Tridiagonal band, spacing $d=0.1$. \\
\verb|\mbDiagBandN{n}|
  & $n$-diagonal band ($n$ odd), fixed step $0.1$. \\
\verb|\mbBidiagUpper|
  & Main diagonal + super-diagonal ($d=0.1$). \\
\verb|\mbBidiagLower|
  & Main diagonal + sub-diagonal ($d=0.1$). \\
\verb|\mbHessenberg|
  & Upper-triangular outline + sub-diagonal. \\
\verb|\mbHessenberg[color]|
  & Same with filled interior. \\
\midrule
\verb|\mbDiagFrom{x1}{y1}{x2}{y2}|
  & Main diagonal, from $(x_1,y_1)$ to $(x_2,y_2)$ \\
\verb|\mbDiagBandFrom{x1}{y1}{x2}{y2}|
  & Tridiagonal band, clipped. \\
\verb|\mbDiagBandNFrom{n}{x1}{y1}{x2}{y2}|
  & $n$-diagonal band, clipped. \\
\midrule
\verb|\mbFill[col]{x0}{y0}{x1}{y1}|
  & Filled rectangle. Default color \texttt{gray!20}. \\
\verb|\mbVLine{x}|
  & Vertical line, full height $0$ to $H$. \\
\verb|\mbHLine{y}|
  & Horizontal line, full width. \\
\midrule
\verb|\mbLabel{x}{y}{text}|
  & Math-mode label at $(x,y)$. \\
\verb|\mbLabel[color]{x}{y}{text}|
  & Same with filled node label\\
\verb|\mbLabelRaw{x}{y}{text}|
  & Raw-text label at $(x,y)$. \\
\verb|\mbLabelCenter{text}|
  & Math label at center $(0.5,\,H/2)$, any aspect. \\
\midrule
\verb|\mbSpike{x}{y}{dir}|
  & L-corner; \texttt{dir} $\in$ \{\texttt{tl, tr, bl, br}\}. \\
\verb|\mbBlank|
  & Empty content (brackets only). \\
\verb|\mbH|
  & Expands to current aspect value inside content. \\
\bottomrule
\end{tabular}
\end{center}
